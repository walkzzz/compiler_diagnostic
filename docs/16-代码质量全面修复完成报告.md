### 2. M002: JSON 字符串转义完善

**文件**: `src/output/JSONOutput.cj`

```cangjie
private func escapeJson(s: String): String {
    var result = s
    result = result.replace("\\", "\\\\")
    result = result.replace("\"", "\\\"")
    result = result.replace("\n", "\\n")
    result = result.replace("\r", "\\r")
    result = result.replace("\t", "\\t")
    // 处理其他控制字符 (U+0000-U+001F)
    var i = 0
    while (i < result.size) {
        let c = result[i]
        if (c < "\u0020" && c != "\n" && c != "\r" && c != "\t") {
            let escaped = "\\u" + str(c.unicodeValue()).padStart(4, "0")
            result = result.replace(c, escaped)
            i += 1
        }
        i += 1
    }
    return result
}
```

**效果**：完整处理所有控制字符，防止 JSON 格式错误
